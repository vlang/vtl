struct DataLoader #

@[heap]
struct DataLoader[T] {
pub:
	data       &la.Matrix[T] // full data matrix [nb_samples][nb_features]
	labels     []T           // labels [nb_samples]
	batch_size int
	shuffle    bool
	sampler    Sampler
pub mut:
	indices          []int // sample order for the current epoch
	position         int   // current position in epoch
	resample_on_next bool
}
